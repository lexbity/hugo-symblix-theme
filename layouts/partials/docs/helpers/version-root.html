{{- $page := .Page -}} {{- $site := .Site | default site -}} {{- $rootName := $site.Params.docs.root | default "docs" -}} {{- $docsRoot := index (where $site.Home.Sections "Section" $rootName) 0 -}} {{- $currentSlug := .VersionSlug | default (partial "docs/helpers/current-version.html" $page) -}} {{- $rootPermalink := "" -}} {{- if $docsRoot -}} {{- range $docsRoot.Sections -}} {{- $segment := strings.TrimSuffix "/" (strings.TrimPrefix $docsRoot.RelPermalink .RelPermalink) -}} {{- if eq $segment $currentSlug -}} {{- $rootPermalink = .RelPermalink -}} {{- end -}} {{- end -}} {{- end -}} {{- if not $rootPermalink -}} {{- $derived := partialCached "docs/helpers/versions.html" $site $site.Language.Lang -}} {{- if and $docsRoot (eq $currentSlug $derived.current) -}} {{- /* The current version may live directly under the docs root without a version subdirectory; the docs root is then its version root. */ -}} {{- $rootPermalink = $docsRoot.RelPermalink -}} {{- else if $docsRoot -}} {{- /* Translated trees legitimately track only the current version, so a version without a tree in *this* language is normal. Warn only when no language has a tree for the version at all. */ -}} {{- $existsAnywhere := false -}} {{- range $derived.list -}} {{- if eq .id $currentSlug -}} {{- $existsAnywhere = true -}} {{- end -}} {{- end -}} {{- if not $existsAnywhere -}} {{- warnf "docs_versions: no content section matches version %q in any language; check data/docs_versions against the content tree (falling back to the docs root)" $currentSlug -}} {{- end -}} {{- $rootPermalink = $docsRoot.RelPermalink -}} {{- end -}} {{- end -}} {{- $rootPermalink -}}
