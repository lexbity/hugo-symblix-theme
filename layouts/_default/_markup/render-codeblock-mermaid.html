{{- /* Mermaid code blocks render at build time through the diagram pipeline (scripts/render-diagrams.mjs), which hashes the trimmed block source and writes assets/diagrams/.svg (light) and .dark.svg (dark theme). The hook serves those SVGs as static images, so published pages stay JavaScript-free. A missing diagram falls back to a plain code block; set params.docs.mermaid.enforce to warn (fails --panicOnWarning builds) when a diagram was not pre-rendered. */ -}} {{- $src := replaceRE "\r\n" "\n" .Inner | strings.TrimSpace -}} {{- $hash := sha256 $src -}} {{- $light := "" -}} {{- with resources.Get (printf "diagrams/%s.svg" $hash) -}} {{- $light = . | minify -}} {{- end -}} {{- $dark := "" -}} {{- with resources.Get (printf "diagrams/%s.dark.svg" $hash) -}} {{- $dark = . | minify -}} {{- end -}} {{- $title := "" -}} {{- with .Attributes.title -}} {{- $title = . -}} {{- end -}} {{- if $light -}} {{- $alt := "" -}} {{- with .Attributes.alt -}} {{- $alt = . -}} {{- end -}} {{- $alt = or $alt (or $title (i18n "diagram" | default "Diagram")) }} {{- else -}} {{- with site.Params.docs.mermaid.enforce -}} {{- warnf "mermaid block %s has no pre-rendered diagram; run the diagram pipeline (npm run diagrams)" $hash -}} {{- end -}} {{- end -}}
